## 3. 执行记录与结果 <a id="run"></a>

```bash
# 核心三包（typecheck / lint / test）
cd bms && npm run check

# 宿主（PC）
cd frontend/apps/desktop && npx vue-tsc -b && npm run lint && npm run test:cov && npm run build && npm run budget

# 模块工程（远端产物 + 元数据 + 隔离护栏）
cd frontend/modules/demo && npm run lint && npm run typecheck && npm run test && npm run build && npm run budget && npm run guard:isolation

# 平台脚本（本地与 CI 同口径）
cd bms && node frontend/scripts/check-module-isolation.mjs
node frontend/scripts/check-shared-whitelist.mjs
node frontend/scripts/check-module-manifest.mjs

# 真实发布（演示模块）
node frontend/scripts/release-module.mjs publish --module demo --by minjian
node frontend/scripts/release-module.mjs status --module demo

# 回滚端到端演练（临时工作区；构建 → 发布 0.1.0 → 升 0.2.0 发布 → 回滚 0.1.0）
node frontend/scripts/release-module.mjs publish  --module demo --root /tmp/opencode/03_02-drill/frontend --origin http://localhost:5012 --by drill
node frontend/scripts/release-module.mjs publish  --module demo --root /tmp/opencode/03_02-drill/frontend --origin http://localhost:5012 --by drill   # 版本升 0.2.0 后
node frontend/scripts/release-module.mjs rollback --module demo --root /tmp/opencode/03_02-drill/frontend --by drill
node frontend/scripts/release-module.mjs disable  --module demo --root /tmp/opencode/03_02-drill/frontend --by drill
node frontend/scripts/release-module.mjs enable   --module demo --root /tmp/opencode/03_02-drill/frontend --by drill
node frontend/scripts/serve-module-releases.mjs --root /tmp/opencode/03_02-drill/frontend --port 5012   # 发布存储（演练）
# 归档容器加载核对（Node 直接加载归档产物暴露的模块定义）
node --input-type=module -e "…import('file:///…/releases/demo/0.1.0/remoteEntry.js')…"

# 文档与基座校验
python3 scripts/tools/base-check/check-base.py
python3 scripts/tools/base-check/check-links.py
python3 scripts/tools/check-docs/check-status.py --stage 05_前端插件化 --strict
```
